#include <stdio.h>
#include <stdlib.h>

void strassenMatrixMultiply(int a[2][2], int b[2][2], int result[2][2]) {
    int p1 = (a[0][0] + a[1][1]) * (b[0][0] + b[1][1]);
    int p2 = (a[1][0] + a[1][1]) * b[0][0];
    int p3 = a[0][0] * (b[0][1] - b[1][1]);
    int p4 = a[1][1] * (b[1][0] - b[0][0]);
    int p5 = (a[0][0] + a[0][1]) * b[1][1];
    int p6 = (a[1][0] - a[0][0]) * (b[0][0] + b[0][1]);
    int p7 = (a[0][1] - a[1][1]) * (b[1][0] + b[1][1]);

    result[0][0] = p1 + p4 - p5 + p7;
    result[0][1] = p3 + p5;
    result[1][0] = p2 + p4;
    result[1][1] = p1 + p3 - p2 + p6;
}

void inputMatrix(int matrix[4][4]) {
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            scanf("%d", &matrix[i][j]);
        }
    }
}

void divideMatrix(int matrix[4][4], int submatrix[2][2], int row, int col) {
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            submatrix[i][j] = matrix[row + i][col + j];
        }
    }
}

void combineSubmatrices(int c[4][4], int result[2][2], int row, int col) {
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            c[row + i][col + j] = result[i][j];
        }
    }
}

void strassenMultiply(int a[4][4], int b[4][4], int c[4][4]) {
   int a11[2][2], a12[2][2], a21[2][2], a22[2][2];
    int b11[2][2], b12[2][2], b21[2][2], b22[2][2];
    int c11[2][2], c12[2][2], c21[2][2], c22[2][2];

    divideMatrix(a, a11, 0, 0);
    divideMatrix(a, a12, 0, 2);
    divideMatrix(a, a21, 2, 0);
    divideMatrix(a, a22, 2, 2);

    divideMatrix(b, b11, 0, 0);
    divideMatrix(b, b12, 0, 2);
    divideMatrix(b, b21, 2, 0);
    divideMatrix(b, b22, 2, 2);

    strassenMatrixMultiply(a11, b11, c11);
    strassenMatrixMultiply(a12, b12, c12);
    strassenMatrixMultiply(a21, b21, c21);
    strassenMatrixMultiply(a22, b22, c22);

    combineSubmatrices(c, c11, 0, 0);
    combineSubmatrices(c, c12, 0, 2);
    combineSubmatrices(c, c21, 2, 0);
    combineSubmatrices(c, c22, 2, 2);
}

int main() {
    int a[4][4], b[4][4], c[4][4] = {0};

    inputMatrix(a);
    inputMatrix(b);

    strassenMultiply(a, b, c);

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }

    return 0;
}
