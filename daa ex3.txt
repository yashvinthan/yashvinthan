#include <iostream>
using namespace std;

void warshall(int graph[][100], int num_nodes) {
    for (int k = 0; k < num_nodes; k++) {
        for (int i = 0; i < num_nodes; i++) {
            for (int j = 0; j < num_nodes; j++) {
                if (graph[i][k] && graph[k][j]) {
                    graph[i][j] = 1;
                }
            }
        }
    }

    for (int i = 0; i < num_nodes; i++) {
        for (int j = 0; j < num_nodes; j++) {
            cout << graph[i][j];
            if (j < num_nodes - 1)
                cout << " ";
        }
        cout << "\n";
    }
}

int main() {
    int num_nodes;
    cin >> num_nodes;
    int graph[100][100];

    for (int i = 0; i < num_nodes; i++) {
        for (int j = 0; j < num_nodes; j++) {
            cin >> graph[i][j];
            if (graph[i][j] != 0) {
                graph[i][j] = 1;
            }
        }
    }

    warshall(graph, num_nodes);
    return 0;
}
